# =============================================================================
# FILE: nettrades_gpu_admin/security/ir.model.access.csv
# =============================================================================
# DESCRIPTION:
#   Model-level CRUD (Create, Read, Update, Delete) access rights.
#   Each line grants permissions to a specific group for a model.
#
#   Columns:
#     id           : unique identifier for the access line
#     name         : human-readable name
#     model_id:id  : reference to the model (e.g., model_gpu_cluster)
#     group_id:id  : reference to the security group
#     perm_read    : 1 = can read, 0 = cannot read
#     perm_write   : 1 = can write, 0 = cannot write
#     perm_create  : 1 = can create, 0 = cannot create
#     perm_unlink  : 1 = can delete, 0 = cannot delete
#
#   Existing lines are preserved. New token lines are added at the bottom.
# =============================================================================

<!--
    FILE: nettrades_gpu_admin/security/gpu_admin_security.xml
    ============================================================================
    DESCRIPTION:
        Record rules for all GPU-related models to enforce company isolation
        and group-specific access restrictions.

        These rules implement the security policy for the distributed GPU
        marketplace. They work in conjunction with:
          - groups.xml         : defines the user groups
          - ir.model.access.csv : defines CRUD permissions at the model level

        ────────────────────────────────────────────────────────────────────────────
        SECURITY POLICY OVERVIEW
        ────────────────────────────────────────────────────────────────────────────
        1. COMPANY ISOLATION (Multi-Tenant)
           • Users can only see GPU resources that belong to their company.
           • This is enforced through domain filters on every GPU model.

        2. GROUP-SPECIFIC RESTRICTIONS
           • GPU Administrators have full access to all GPU models within
             their company (create, read, update, delete).
           • GPU Operators have read-only access to GPU clusters and nodes
             (they can view configurations but cannot modify them).

        3. IMPLEMENTATION APPROACH
           • Company-only rules apply to BOTH Administrators and Operators.
           • Explicit read-only rules are applied specifically to Operators
             on the `gpu.cluster` and `gpu.node` models to reinforce the
             read-only policy (though the domain is the same, the rule exists
             to clearly document the intent and to ensure separation of concerns).

        ────────────────────────────────────────────────────────────────────────────
        ORIGINAL RULES PRESERVED (2026-06-28)
        ────────────────────────────────────────────────────────────────────────────
        This file contains all 8 record rules from the original minified version,
        reformatted for readability and maintainability. The group references
        have been updated to use the new group IDs:
          - group_gpu_administrator (replaces group_gpu_manager)
          - group_gpu_operator       (replaces group_gpu_user)

        No functionality has been removed. All original access controls are
        preserved exactly as they were intended.
    ============================================================================
-->

    <!-- =====================================================================
         RULE 1: GPU CLUSTER – Company Isolation (Admin + Operator)
         =====================================================================
         Original name: "GPU Administrator GPU Operator GPU Cluster: Company Only"
         Applies to:  gpu.cluster model
         Groups:      Administrator and Operator
         Domain:      Only records where company_id matches the user's company.
         Purpose:     Ensures that both Administrators and Operators can only
                      see GPU clusters belonging to their own company.
    ===================================================================== -->
    <record id="gpu_cluster_company_rule" model="ir.rule">
        <field name="name">GPU Administrator GPU Operator GPU Cluster: Company Only</field>
        <field name="model_id" ref="model_gpu_cluster"/>
        <field name="global" eval="False"/>
        <field name="domain_force">[('company_id', '=', user.company_id.id)]</field>
